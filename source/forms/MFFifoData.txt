Private Sub Valori()
Dim sRealized As Double
Dim sRealizedE As Double
Dim sCambio As Double
Dim sProdotto As String





Set db = CurrentDb
    If Not IsNull(Productid.Column(0)) Then
        sProdotto = "='" & Productid.Column(0)
    Else
        sProdotto = "Like '*"
    End If
    
sqlc = "SELECT [1ISIN].ISIN, Change.[C_LAST_PRICE_REF ( LIVEQUOTE )] as Cambio " & _
        "FROM Change INNER JOIN 1ISIN ON Change.IdC = [1ISIN].IdC " & _
        "WHERE ((([1ISIN].ISIN)" & sProdotto & "'));"



    Set rsR = db.OpenRecordset(sqlc, dbOpenSnapshot)
    
'pulisco le variabili
        sCambio = 0
        sRealized = 0
        sRealizedE = 0
        
'Apro il case per le varie opzioni

        If Not rsR.EOF Then
            sCambio = rsR("Cambio")
        Else
            sCambio = 1
            
        
        End If


If TValue = "" Then
    sRealized = 0
Else
    If TValue = 0 Then
    
     '  If TNetP Is Null Then
     '       TNetP = 0
            
        sRealized = TNetS - TNetP
        sRealizedE = sRealized / sCambio
    '    End If
    Else
    '   If Not IsNull(TNetP) Or TNetP = "" Then
    '        TNetP = TNetP
    '        Else
    '        TNetP = 0
    '    End If
            
        sRealized = TValue + TNetS - TNetP
        sRealizedE = TValue / sCambio
    End If
    
End If
'srealized=

txtRealized = sRealized
'IIf(IsError([SaleDetail].[Form]![TNetS]);[TValue];([TValue]-([TNetP]-[TNetS])))


txtValueE = sRealizedE
'=IIf(IsError([fifoStock].[Form]![CValue]);0;[fifoStock].[Form]![CValue])



End Sub
Private Sub CaricaIsin()

Set db = CurrentDb



sqlc = "SELECT [1ISIN].ISIN, [1ISIN].SYM, [1ISIN].IdS, [1ISIN].IdA, " & _
            "[1ISIN].IdD, [1ISIN].IdT, [1ISIN].IdR, [1ISIN].IdC " & _
        "FROM 1ISIN INNER JOIN [1AKJ trade report] ON [1ISIN].ISIN = [1AKJ trade report].ISIN " & _
        "WHERE ((([1AKJ trade report].TD) <= #" & Format(sDataCerca, "yyyy-mm-dd") & "#)) " & _
