Private Sub cmdEsci_Click()
On Error GoTo Err_cmdEsci_Click


    If Me.Dirty Then Me.Dirty = False
    DoCmd.Close

Exit_cmdEsci_Click:
    Exit Sub

Err_cmdEsci_Click:
    MsgBox Err.Description
    Resume Exit_cmdEsci_Click
    
End Sub

Private Sub ISIN_DblClick(Cancel As Integer)

On Error GoTo Err_ISIN_DblClick

    Dim stDocName As String
    Dim stLinkCriteria As String

    stDocName = "MISINModify"
    stLinkCriteria = "[ISIN]=" & "'" & Me![ISIN] & "'"
    DoCmd.OpenForm stDocName, , , stLinkCriteria

Exit_ISIN_DblClick:
    Exit Sub

Err_ISIN_DblClick:
    MsgBox Err.Description
    Resume Exit_ISIN_DblClick
    

End Sub
