Private Sub Controllo()
Dim TDol As Double
Dim TEur As Double
Dim TGb As Double
Dim TYen As Double

'pulisco le variabili
    txtDol = 0
    txtEur = 0
    txtGb = 0
    txtYen = 0

    txtDolG = 0
    txtEurG = 0
    txtGbG = 0
    txtYenG = 0

    txtDolA = 0
    txtEurA = 0
    txtGbA = 0
    txtYenA = 0
    
    TDol = 0
    TEur = 0
    TGb = 0
    TYen = 0
    
    
'Inserisco le variabili per la scrittura nella maschera
 '   txtDol = Format$(sDol, "$ #.##0,#")
 '   txtEur = Format$(sEur, "€ #.##0,#")
 '   txtGb = Format$(sGb, "£ #.##0,#")
 '   txtYen = Format$(sYen, "¥ #.##0,#")
'    txtDol = Val(sDol)
'    txtEur = Val(sEur)
'    txtGb = Val(sGb)
'    txtYen = Val(sYen)
'
'
'    txtDolG = Val(sDolG)
'    txtEurG = Val(sEurG)
'    TxtGbG = Val(sGbG)
'    txtYenG = Val(sYenG)
'
'
'    txtDolA = Val(sDolA)
'    txtEurA = Val(sEurA)
'    txtGbA = Val(sGbA)
'    txtYenA = Val(sYenA)'
'
'    txtDolT = Val(sDol) + Val(sDolG) - Val(sDolA)
'    txtEurT = Val(sEur) + Val(sEurG) - Val(sEurA)
'    txtGbT = Val(sGb) + Val(sGbG) - Val(sGbA)
'    txtYenT = Val(sYen) + Val(sYenG) - Val(sYenA)
'
    txtDol = sDol
    txtEur = sEur
    txtGb = sGb
    txtYen = sYen
      
    
    txtDolG = sDolG
    txtEurG = sEurG
    txtGbG = sGbG
    txtYenG = sYenG
        
 
    txtDolA = sDolA
    txtEurA = sEurA
    txtGbA = sGbA
    txtYenA = sYenA

    TDol = sDol + sDolG - sDolA
    TEur = sEur + sEurG - sEurA
    TGb = sGb + sGbG - sGbA
    TYen = sYen + sYenG - sYenA
    
    txtDolT = TDol
    txtEurT = TEur
    txtGbT = TGb
    txtYenT = TYen
    
    
    txtDolTE = TDol / sDolT
    txtEurTE = TEur / 1
    txtGbTE = TGb / sGbT
    txtYenTE = TYen / sYenT
   
    txtTotal = txtDolTE + txtEurTE + txtGbTE + txtYenTE
    txtTotalLiveE = sTotaleLiveE

End Sub
